### 1.2 Кромка в шейпе

```js
// src/modules/shape/schema.js:6
var SHAPE_EDGE_OPS = ['Rough Arris','Flat Polish','CNC Shape Polish','Mitering','Beveling'];

// src/modules/shape/features.js:83
function shapeEdgeOps(def,id){ return (def.edgeOps && def.edgeOps[id]) || []; }
```

`def.edgeOps` — объект `{ <edgeId>: [ {type, angle?, width?, side?}, … ] }`, нормализуется в `schema.js:127-134`.

### 1.3 Припуск по толщине — УЖЕ ЕСТЬ

```js
// src/modules/shape/features.js:80-87
function shapeThicknessMm(def){ … Number(def.thickness) … }
function shapePolishAllowance(th){ if(th>=3&&th<=6) return 1/16; if(th>=8&&th<=10) return 1/8; if(th>=12&&th<=15) return 3/16; return 0; }
function shapeOperationAllowance(type,th){
  if(type==='CNC Shape Polish') return .25;
  if(type==='Flat Polish'||type==='Beveling'||type==='Mitering') return shapePolishAllowance(th);
  return 0;                                   // Rough Arris размер не меняет
}
function shapeEdgeAllowance(def,edge){ … Math.max(...ops.map(op=>shapeOperationAllowance(op.type,th))) … }
```

И это **доходит до размера реза**: `src/modules/shape/cutting.js:27` строит offset контура по `shapeEdgeAllowance` на каждое ребро.

**Почему пользователь этого не видит.** Толщина берётся из `def.thickness` — текстового поля самого шейпа, у которого дефолт `'6'` (`schema.js:133`). Она **никак не связана с Makeup строки**. Значит рейлинг 12 мм получает припуск по бэнду 6 мм, если оператор не вспомнил вписать 12 в шейп руками.

> **Это и есть корень проблемы «нет бордера по толщине». Не строить систему припусков заново — связать `shape.thickness` с толщиной Makeup строки.**

## 3. Решение: Service Set

**Обработка кромки — свойство строки заказа, а не геометрии.** Вводится сущность уровня заказа, устроенная как существующий Makeup: задаётся один раз, назначается на любое число строк.

### 3.1 Данные

В `normalizeSalesOrder` (`src/erp/sales/data.js:126`) добавляется `serviceSets: []`:

```js
serviceSet = {
  id: 'SVC-…',                 // salesEntityId
  code: 'S1',                  // короткий код, уникален внутри заказа
  name: 'Seam all round',
  mode: 'sides' | 'perimeter', // РОВНО ДВА режима, третьего нет — см. 3.4
  sides: { A:[op], B:[op], C:[op], D:[op], other:[op] },
  perimeter: [op],
  featureServices: { hole:'…', notch:'…', cutout:'…' },
  createdAt, updatedAt
}
```

`op` — объект того же вида, что в `shapeDef.edgeOps`: `{type, angle?, width?, side?}`, `type` из `SHAPE_EDGE_OPS`. Переиспользовать `shapeNormalizeOp` из `schema.js:15`, не заводить второй словарь операций.

В `normalizeSalesOrderLine` (`data.js:121`) добавляются:

```js
serviceSetId: '',                    // ссылка на набор заказа, пусто = не назначен
serviceOverrides: {                  // ручные правки, ключ — ФИЗИЧЕСКОЕ РЕБРО
  pinnedRevision: 2,                 // ревизия шейпа, против которой правили
  edges: { 'A':[op], 'N2-b':[op] }
},
features: { hole:0, notch:0, cutout:0 }   // только для DXF, у шейпа берётся из геометрии
```

**Количество отверстий хранится на строке, а не в наборе.** Набор говорит, ЧЕМ обрабатывать, геометрия — СКОЛЬКО их. Иначе все строки с одним набором получат одинаковое число отверстий, чего не бывает.

### 3.2 Резолвер — центральная точка

Приоритет: **override строки → набор строки → `shapeDef.edgeOps`**.

```js
// НОВОЕ, в src/erp/sales/service-set.js
function salesResolveEdgeOps(order, line, shapeDef, edgeId){
  const ovr = line.serviceOverrides && line.serviceOverrides.edges;
  if (ovr && ovr[edgeId]) return ovr[edgeId];
  const set = salesServiceSetById(order, line.serviceSetId);
  if (set){
    if (set.mode === 'perimeter') return set.perimeter;
    const key = ['A','B','C','D'].includes(edgeId) ? edgeId : 'other';
    return set.sides[key] || [];
  }
  return shapeEdgeOps(shapeDef, edgeId);   // старое поведение как fallback
}
```

**Функцию `shapeEdgeOps` в `src/modules/shape/features.js` НЕ трогать.** Она в модуле, который не имеет права знать про заказы. Резолвер живёт в слое `erp/sales`.

**Единственная правка в биллинге** — `orders.js:170`, замена одного вызова:

```js
// было
(shapeEdgeOps(s, g.id) || []).forEach(function(op){ … });
// стало
(salesResolveEdgeOps(soDraft, line, s, g.id) || []).forEach(function(op){ … });
```

После этого ставки, модалки сервисов, итоги заказа и «без цены» продолжают работать без изменений — они просто начинают видеть кромку, которую никто не проставлял руками.

**Последствие, которое надо решить осознанно.** `shapeEdgeAllowance` и `shapeProductionRequirements` в `features.js` тоже читают `shapeEdgeOps` и Service Set не увидят. Значит панель «Производственные требования» и **offset контура резки** останутся на данных шейпа. Два варианта:

- **(A) рекомендуемый:** при открытии шейпа из строки (`salesOrderConfigureShape`, `orders.js`) слой sales **материализует** резолв в `sDraft.edgeOps` — то есть набор физически проставляет галочки в матрице кромок, а дальше всё работает как раньше. Пользователь видит проставленную кромку, модуль остаётся неведающим, резка считается верно. Ручная правка после этого = override.
- (B) добавить модулю необязательный вход `resolvedEdgeOps` и передавать его в `ShapeModule.compute`. Чище архитектурно, но меняет публичный контракт модуля и все его тесты.

Взять (A), если пользователь не скажет иначе.

### 3.3 Толщина — закрыть проблему №2

При материализации (вариант A) и при сохранении строки прописывать в шейп толщину из Makeup:

```js
shapeDef.thickness = String(salesLineGlassThicknessMm(line));
```

Толщину строки уже умеет считать `salesPricingThickness(line)` (`orders.js:160`) — переиспользовать.
Если у строки лайты **разной** толщины (`v.length !== 1`), толщина не определена: строка получает статус `Needs thickness`, припуск не считается, молчаливая подстановка `'6'` запрещена.

### 3.4 DXF: почему режимов два, а не три

Режим `'both'` (и стороны, и периметр одновременно) **не делать**. Он даёт двойное начисление: ребро получает услугу из `sides` и ту же услугу из `perimeter`, припуск считается дважды, погонные дюймы биллятся дважды.

Правильное решение — не третий режим набора, а **сопоставление сторон на строке**:

```js
line.sideMap = { A:['seg1','seg7'], B:['seg2'], … } | null
```

Пока `sideMap` пуст, DXF-строка с набором в режиме `sides` получает статус `Needs side mapping`. Набор в режиме `perimeter` работает всегда, но требует длины периметра.

Для `perimeter` строке нужна `line.perimeterIn` — если из `def.source.preview` периметр не выводится, оператор вводит его руками, иначе погонные дюймы не посчитать.

### 4.1 Колонка в таблице строк

В `.sales-lines-table` добавляется колонка `Set` сразу после `MU`. В ячейке — короткий бейдж `S2`. Если у строки есть ручные правки — `S2*` другим цветом, с `title`, перечисляющим переопределённые рёбра. Пусто — `—`.

### 4.2 Матрица кромок в модалке

Переиспользовать существующие классы редактора кромок шейпа, не изобретать свои:
`.shape-edgework-scroll`, `.shape-edgework-matrix`, `.shape-edgework-row`, `.shape-edgework-labels`, `.shape-edge-code`, `.shape-op-compact`.
Сетка строки: `minmax(62px,1.15fr) repeat(5,minmax(0,1fr))` — **ровно 5 колонок операций**, потому что `SHAPE_EDGE_OPS` содержит 5 элементов. Подписи в шапке: `Rough · Flat · CNC · Miter · Bevel`.

Строки матрицы: `A` (Left · height), `B` (Bottom · width), `C` (Right · height), `D` (Top · width), `OTHER` (notch & elbow ribs). В режиме `perimeter` — одна строка `ALL`.

Пресеты над матрицей, теми словами, которыми говорит цех:
`все 4 стороны` · `по ширине (B+D)` · `по высоте (A+C)` · `очистить`.

### 4.3 Наборы должны различаться с одного взгляда

Пользователь не смог отличить S1 от S2, потому что чип показывал только имя. На каждом наборе — **формула**, а не название:

```
S1   ABCD Rough                    1 строка
S2   AC CNC · BD Rough             6 строк
S3   контур CNC                    0 строк
```

Формула строится группировкой сторон с одинаковым списком операций, короткие имена: `Rough`, `Flat`, `CNC`, `Miter`, `Bevel`.

### 4.4 Фильтры над таблицей строк

Ряд компактных кнопок со счётчиками сразу под заголовком `ORDER LINES`, до bulk-бара:

```
Все 40 · Без набора 33 · Требуют внимания 1 · С ручными правками 2 · DXF 2
```

«Без набора» и «Требуют внимания» подсвечиваются янтарным, пока в них есть строки.
**Галочка «выделить всё» в шапке таблицы работает по текущему фильтру**, а не по всему заказу — это главный смысл фильтров: отфильтровал проблемные, выделил одной галкой, исправил.

### 4.5 Bulk-бар: политика и предпросмотр

В `.sales-bulk` (`sales.js:36`) добавляются:
- селект `Assign Service Set`;
- селект `Политика`: `сохранить ручные правки` (по умолчанию) · `перезаписать всё` · `только пустые`;
- кнопка `Показать что изменится`.

Кнопка не применяет, а показывает панель предпросмотра:


И только потом «Применить к N строкам» / «Отмена». На заказе в 100 строк это единственная защита от массовой ошибки.

**Правку CSS не забыть:** `.sales-bulk select{min-width:260px}` в `src/styles/modules.css` — при трёх селектах бар не влезает. Нужно `flex-wrap:wrap` и `min-width` порядка 150px, ограничив селектором `.sales-lines-block .sales-bulk`, чтобы не задеть другие места.

### 4.6 Модалка кромок строки

Кнопка в строке открывает матрицу физических рёбер именно этой строки. Каждое ребро подписано источником: `из S2` серым или `ручная правка` цветом override, с кнопкой `вернуть к набору`. Внизу — `вернуть всю строку к набору`.

Разметка — `.sales-service-modal-back` → `.sales-service-modal.line-modal`, как у `salesLineServicesModal()`.

### 4.7 Пин ревизии

Если `serviceOverrides.pinnedRevision !== shapeRef.revision` и среди переопределённых рёбер есть те, которых больше нет в контуре — строка получает статус ошибки, а в модалке кромок выводится предупреждение с кнопкой `удалить потерянные`. **Молча выбрасывать нельзя, решает человек.**

### 4.8 Excel-импорт

`salesExcelApply()` (`orders.js:101`) получает третий режим колонок: `MU | SS | Qty | Width | Height | Mark`. Тогда набор приезжает вместе со строками и назначать вручную не нужно вообще.

## 5. Статусы строки

Считаются в одном месте, показываются в таблице и используются фильтром «Требуют внимания».

| Статус | Условие |
|---|---|
| `Not assigned` | `serviceSetId` пуст |
| `Needs perimeter` | набор `perimeter`, источник DXF, `perimeterIn` пуст |
| `Needs side mapping` | набор `sides`, источник DXF, `sideMap` пуст |
| `Needs thickness` | `salesPricingThickness(line).ok === false` |
| `Override lost edge` | пин ревизии разошёлся и ребро исчезло (4.7) |
| `Unassigned edges` | набор `sides`, у части физических рёбер пустой список операций |
| `Has override` | есть ручные правки, всё остальное в порядке |
| `Ready` | ничего из перечисленного |

## 6. Открытые вопросы — задать пользователю ДО кода

1. **`Rough Arris` + `Flat Polish` на одном ребре — последовательные операции или взаимоисключающие?**
   Это не теория. В коде уже два разных ответа:
   - припуск: `shapeEdgeAllowance` берёт `Math.max` (`features.js:87`), то есть считает их совместимыми и берёт больший;
   - деньги: `salesLineChargeRows` (`orders.js:170`) заводит **две отдельные строки начисления** и биллит обе.

   Если в цеху это «либо-либо», то в текущем коде **уже есть тихая переплата**, и её надо чинить отдельным коммитом независимо от Service Set. Если это две последовательные операции — всё верно, но правило надо объявить явно, а не выводить из кода.

2. **Одинаковая ли кромка у всех лайтов стеклопакета?** Предлагаемая модель: набор действует на всю строку. Если бывает наружный полированный при внутреннем под seam — структура меняется, набор должен адресоваться на лайт.

3. **Митра и фацет: припуск идёт в ширину, в высоту или в обе?** Предлагаемое правило: припуск идёт по оси, перпендикулярной стороне. Подтвердить.

4. Куда деть `Mitering 45°` и `Beveling` **без ставок в каталоге** — они дают «без цены» уже сегодня.

## 7. Миграция существующих данных

1. Заказы без `serviceSets` получают пустой массив, строки — пустой `serviceSetId`. Резолвер падает в `shapeEdgeOps`, поведение не меняется. **Ни один существующий заказ не должен посчитаться иначе после этой ревизии** — это критерий приёмки.
2. `shapeDef.edgeOps` **не удалять и не переносить принудительно.** Он остаётся источником по умолчанию.
3. Опционально, отдельной кнопкой в модалке: «собрать набор из этой строки» — читает резолвнутую кромку строки и создаёт из неё Service Set.

## 9. Тесты, которые обязаны появиться в `test/run.js`

1. Резолвер: override побеждает набор, набор побеждает `shapeDef.edgeOps`, пустой набор падает в `edgeOps`.
2. Заказ без `serviceSets` даёт **точно те же** `salesLineChargeRows`, что до ревизии.
3. Пин ревизии: изменение `shapeRef.revision` с исчезновением ребра даёт статус ошибки, а не тихую потерю.
4. Политика назначения: `keep` сохраняет overrides, `overwrite` их чистит, `empty` не трогает строки с уже назначенным набором.
5. Припуск: одно ребро `CNC Shape Polish` при 12 мм даёт 1/4″; `Flat Polish` при 12 мм даёт 3/16″; `Rough Arris` даёт 0.
6. `Needs thickness`: строка с лайтами разной толщины не получает припуск и не подставляет 6 мм.
7. Excel-импорт в режиме с колонкой SS проставляет набор.
8. Уникальность `code` набора внутри заказа, отсутствие дублей `id`.

Полный набор должен остаться зелёным: `node test/run.js`.
